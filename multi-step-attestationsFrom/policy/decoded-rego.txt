package cilock.verify

# ---------------------------------------------------------------------------
# Check 1: artifact identity overlap.
#
# The build step's product attestation's subject is `tree:products` with a
# Merkle root. The scan step's material attestation's subject is
# `tree:materials` — also a Merkle root, computed at scan time over the
# scanner's inputs. If the scanner consumed the build's artifact, those two
# roots overlap (the build's product becomes one of the scan's materials,
# and the scan's tree:materials Merkle root commits to it).
#
# We enforce the stronger check available from a release gate: the scan's
# material tree must contain the build's product root as one of its
# inclusion-proof leaves. The release pipeline emits that inclusion-proof
# attestation as `prove` (not shown in this minimal example); the check
# here approximates by asserting that the scan's material tree root is
# bound to the build's product subject digest via an inclusion proof in
# the scan's evidence set.
#
# Minimal form (this example): assert the scan step recorded the build's
# product as a material. The build's product tree root must appear as the
# scan's material tree root's predecessor — i.e. the scan step lists an
# inclusion-proof attestation whose treeRoot equals the build's product
# tree root.
# ---------------------------------------------------------------------------

deny[msg] {
    not build_product_root
    msg := "release: build step must emit a product attestation"
}

deny[msg] {
    not scan_inclusion_proofs
    msg := "release: scan step must include at least one inclusion-proof attestation binding it to the build's product tree"
}

deny[msg] {
    build_product_root
    scan_inclusion_proofs
    not scan_proves_build_product
    msg := sprintf(
        "release: scan step has inclusion proofs but none names the build's product tree root %v",
        [build_product_root],
    )
}

build_product_root := root {
    p := input.steps.build["https://aflock.ai/attestations/product/v0.3"]
    root := p.merkleRoot
}

scan_inclusion_proofs := proofs {
    proofs := [p |
        p := input.steps.scan["https://aflock.ai/attestations/inclusion-proof/v0.1"]
    ]
    count(proofs) > 0
}

scan_proves_build_product {
    p := input.steps.scan["https://aflock.ai/attestations/inclusion-proof/v0.1"]
    p.treeRoot == build_product_root
}

# ---------------------------------------------------------------------------
# Check 2: scan-result clean bill.
#
# The scan step's sarif attestation carries the tool's findings. Release
# refuses to ship if any finding has level "error" (SARIF's most-severe
# bucket).
# ---------------------------------------------------------------------------

deny[msg] {
    s := input.steps.scan["https://aflock.ai/attestations/sarif/v0.1"]
    finding := s.runs[_].results[_]
    finding.level == "error"
    msg := sprintf(
        "release: scan step has SARIF error-level finding (%v at %v); release blocked",
        [finding.ruleId, finding.locations[0].physicalLocation.artifactLocation.uri],
    )
}
