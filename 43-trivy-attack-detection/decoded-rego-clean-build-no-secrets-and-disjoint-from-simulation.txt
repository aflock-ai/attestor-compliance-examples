package cilock.verify

# ---------------------------------------------------------------------------
# Rule 1: the clean-build step's own secretscan output must be empty.
#
# Same shape as the original single-step gate, kept here as the primary
# release-blocking check.
# ---------------------------------------------------------------------------

deny[msg] {
    input.findings != null
    count(input.findings) > 0
    finding := input.findings[_]
    msg := sprintf("Secret detected in clean-build: %s at %s", [finding.ruleId, finding.location])
}

# ---------------------------------------------------------------------------
# Rule 2: the attack-simulation step MUST have produced findings.
#
# This is a calibration assertion. The attack-simulation step's whole
# purpose is to plant credential-stealer behaviour and let the secretscan
# attestor detect it. If the simulation passes with zero findings, the
# detection ruleset has silently broken — the example no longer
# demonstrates what it claims to demonstrate. Catching that here means
# the trivy attack-detection example fails loudly the moment its own
# regression coverage decays.
#
# This invariant is only expressible because attestationsFrom: ["attack-
# simulation"] lifts the simulation step's secretscan attestation into
# input.steps.attack-simulation.<predicate-type>.
# ---------------------------------------------------------------------------

deny[msg] {
    sim := input.steps["attack-simulation"]["https://aflock.ai/attestations/secretscan/v0.1"]
    sim.findings == null
    msg := "calibration failure: attack-simulation step found zero secrets — the detection ruleset has regressed"
}

deny[msg] {
    sim := input.steps["attack-simulation"]["https://aflock.ai/attestations/secretscan/v0.1"]
    sim.findings != null
    count(sim.findings) == 0
    msg := "calibration failure: attack-simulation step's findings array is empty — the detection ruleset has regressed"
}

# ---------------------------------------------------------------------------
# Rule 3: the clean build must not contain ANY pattern the simulation
# detected. This is the structural check that catches a regression in the
# OPPOSITE direction: a real build accidentally leaking the same secret
# patterns as the planted-attack run.
#
# Implemented as a set-disjointness assertion: for every (ruleId, location)
# pair in the clean-build findings, the same pair must NOT appear in the
# simulation findings. (If clean-build is empty per Rule 1, this rule is
# vacuously satisfied.)
# ---------------------------------------------------------------------------

deny[msg] {
    input.findings != null
    sim := input.steps["attack-simulation"]["https://aflock.ai/attestations/secretscan/v0.1"]
    sim.findings != null
    clean_finding := input.findings[_]
    sim_finding := sim.findings[_]
    clean_finding.ruleId == sim_finding.ruleId
    clean_finding.location == sim_finding.location
    msg := sprintf(
        "clean-build leaked an attack-simulation pattern: ruleId=%s at %s",
        [clean_finding.ruleId, clean_finding.location],
    )
}
